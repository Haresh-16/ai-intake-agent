from __future__ import annotations

class PracticeArea(str, Enum):
    TCPA = "TCPA"
    FCRA = "FCRA"
    FDCPA = "FDCPA"
    MASS_ARB = "MASS_ARB"
    UNKNOWN_CONSUMER = "UNKNOWN_CONSUMER"
    WRONG_FIT = "WRONG_FIT"

    ROUTE_TO_ATTORNEY = "route_to_attorney"
    ROUTE_TO_INTAKE_REVIEW = "route_to_intake_review"
    REQUEST_MORE_EVIDENCE = "request_more_evidence"
    WRONG_FIT_CLOSE = "wrong_fit_close"


class DefendantRole(str, Enum):
    CALLER = "caller"
    COLLECTOR = "collector"
    FURNISHER = "furnisher"
    BUREAU = "bureau"
    OTHER = "other"


class SecondaryPracticeArea(BaseModel):
    area: str
    confidence: float = Field(ge=0.0, le=1.0)
    reason: str
    source: Literal["rule_based", "llm_inferred"] = "rule_based"


class Defendant(BaseModel):
    name: str
    role: DefendantRole
    pattern_bank_hit: bool = False

    no_legal_advice_given: bool
    no_outcome_promised: bool
    no_acr_implied: bool
    disclosure_logged: bool


class ProspectInput(BaseModel):
    first_name: str | None = None
    last_name: str | None = None
    email: str | None = None
    phone: str | None = None
    state: str | None = None
    source_channel: str = "web_form"
    consent_status: str = "pending"
    external_ids: dict[str, str] = Field(default_factory=dict)


class IntakeSummary(BaseModel):
    conversation_id: str
    prospect: ProspectInput
    primary_practice_area: PracticeArea
    primary_confidence: float = Field(ge=0.0, le=1.0)
    secondary_practice_areas: list[SecondaryPracticeArea] = Field(default_factory=list)
    fact_pattern: str = Field(min_length=20, max_length=2000)
    defendants: list[Defendant] = Field(default_factory=list)
    evidence_collected: list[str] = Field(default_factory=list)
    evidence_requested: list[str] = Field(default_factory=list)
    evidence_missing: list[str] = Field(default_factory=list)
    cross_practice_flag: bool
    follow_up_question: str
    agent_recommendation: AgentRecommendation
    compliance_check: ComplianceCheck
    agent_version: str
    llm_request_metadata: dict[str, Any] = Field(default_factory=dict)

    @field_validator("fact_pattern")
    @classmethod
    def validate_fact_pattern(cls, value: str) -> str:
        compact = " ".join(value.split())
        if len(compact.split()) < 12:
            raise ValueError("fact_pattern must be sufficiently descriptive")
        return compact

    @field_validator("evidence_missing")
    @classmethod
    def validate_missing_subset(cls, value: list[str], info):
        requested = set(info.data.get("evidence_requested", []))
        if not set(value).issubset(requested):
            raise ValueError("evidence_missing must be subset of evidence_requested")
        return value

from __future__ import annotations

from fastapi import APIRouter, Depends, HTTPException, Request
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.db import get_session
from app.models import Conversation, IntakeSummaryRecord, Message, Prospect
from app.schemas import APIMessageResponse, ConversationDetail, MessageCreate, ProspectInput
from app.services.audit import write_audit
from app.services.followup import build_follow_up_tasks
from app.services.pii import redact_sensitive_text
from app.services.prompt_loader import load_locked_prompt
from app.services.summary_generator import generate_summary

router = APIRouter(prefix="/api/intake", tags=["intake"])

STAGES = [
    "greeting_and_disclosure",
    "consent",
    "open_fact_gathering",
    "silent_classification",
    "targeted_followups",
    "evidence_request",
    "cross_practice_scan",
    "defendant_capture",
    "wrap",
]


def suggested_prompt(stage_index: int) -> str:
    prompts = [
        "Hello, I am an AI intake assistant, not a lawyer. This chat is not legal advice and does not create an attorney-client relationship. Are you comfortable continuing?",
        "Please describe what happened in your own words.",
        "Who contacted you, how did they contact you, and when did this happen?",
        "Please share any documents, screenshots, letters, call logs, or credit-report materials you already have.",
        "A human reviewer will review what was collected. Do you want to add anything else before we wrap up?",
    ]
    return prompts[min(stage_index, len(prompts) - 1)]


async def _get_or_create_conversation(session: AsyncSession, payload: MessageCreate) -> tuple[Prospect, Conversation]:
    if payload.conversation_id:
        conversation = await session.get(Conversation, payload.conversation_id)
        if not conversation:
            raise HTTPException(status_code=404, detail="Conversation not found")
        prospect = await session.get(Prospect, conversation.prospect_id)
        if not prospect:
            raise HTTPException(status_code=404, detail="Prospect not found")
        return prospect, conversation

    prospect = Prospect(
        first_name=payload.prospect.first_name,
        last_name=payload.prospect.last_name,
        email=payload.prospect.email,
        phone=payload.prospect.phone,
        state=payload.prospect.state,
        source_channel=payload.prospect.source_channel,
        consent_status=payload.prospect.consent_status,
        external_ids=payload.prospect.external_ids,
        status="in_intake",
    )
    session.add(prospect)
    await session.flush()

    _, prompt_hash = load_locked_prompt()
    conversation = Conversation(
        prospect_id=prospect.id,
        channel=payload.prospect.source_channel,
        status="active",
        agent_version=prompt_hash[:12],
    )
    session.add(conversation)
    await session.flush()
    return prospect, conversation


@router.post("/messages", response_model=APIMessageResponse)
async def create_message(payload: MessageCreate, request: Request, session: AsyncSession = Depends(get_session)) -> APIMessageResponse:
    prospect, conversation = await _get_or_create_conversation(session, payload)
    clean = redact_sensitive_text(payload.content)
    message = Message(
        conversation_id=conversation.id,
        role=payload.role,
        content=clean,
        tool_calls=None,
        model_metadata={"sanitized": clean != payload.content},
    )
    session.add(message)
    await write_audit(
        session,
        actor=payload.role,
        action="message.received",
        entity_type="conversation",
        entity_id=conversation.id,
        after_state={"content": clean},
        ip=request.client.host if request.client else None,
    )
    await session.commit()

    result = await session.execute(select(Message).where(Message.conversation_id == conversation.id).order_by(Message.created_at.asc()))
    messages = result.scalars().all()
    prospect_turns = len([m for m in messages if m.role == "prospect"])
    stage_index = min(max(prospect_turns - 1, 0), len(STAGES) - 1)
    return APIMessageResponse(
        conversation_id=conversation.id,
        sanitized_content=clean,
        suggested_next_prompt=suggested_prompt(stage_index),
        current_stage=STAGES[stage_index],
    )


@router.post("/conversations/{conversation_id}/finalize")
async def finalize_conversation(conversation_id: str, session: AsyncSession = Depends(get_session)) -> dict:
    conversation = await session.get(Conversation, conversation_id)
    if not conversation:
        raise HTTPException(status_code=404, detail="Conversation not found")
    prospect = await session.get(Prospect, conversation.prospect_id)
    if not prospect:
        raise HTTPException(status_code=404, detail="Prospect not found")

    result = await session.execute(select(Message).where(Message.conversation_id == conversation.id).order_by(Message.created_at.asc()))
    messages = result.scalars().all()
    transcript_text = "\n".join([f"{m.role}: {m.content}" for m in messages])
    if not transcript_text.strip():
        raise HTTPException(status_code=400, detail="Conversation has no messages")

    summary = await generate_summary(
        session=session,
        conversation_id=conversation.id,
        prospect=ProspectInput(
            first_name=prospect.first_name,
            last_name=prospect.last_name,
            email=prospect.email,
            phone=prospect.phone,
            state=prospect.state,
            source_channel=prospect.source_channel,
            consent_status=prospect.consent_status,
            external_ids=prospect.external_ids,
        ),
        transcript_text=transcript_text,
    )

    record = IntakeSummaryRecord(
        conversation_id=conversation.id,
        primary_practice_area=summary.primary_practice_area.value,
        primary_confidence=f"{summary.primary_confidence:.2f}",
        secondary_practice_areas=[item.model_dump() for item in summary.secondary_practice_areas],
        fact_pattern=summary.fact_pattern,
        defendants=[item.model_dump() for item in summary.defendants],
        evidence_collected=summary.evidence_collected,
        evidence_requested=summary.evidence_requested,
        evidence_missing=summary.evidence_missing,
        cross_practice_flag=summary.cross_practice_flag,
        follow_up_question=summary.follow_up_question,
        agent_recommendation=summary.agent_recommendation.value,
        review_status="pending",
        compliance_check=summary.compliance_check.model_dump(),
        agent_version=summary.agent_version,
        raw_summary_json=summary.model_dump(mode="json"),
    )
    session.add(record)

    if summary.evidence_missing:
        for task in build_follow_up_tasks(conversation.id, summary.evidence_missing, conversation.channel):
            session.add(task)

    conversation.status = "complete"
    await write_audit(
        session,
        actor="agent",
        action="summary.generated",
        entity_type="conversation",
        entity_id=conversation.id,
        after_state={
            "primary_practice_area": summary.primary_practice_area.value,
            "agent_version": summary.agent_version,
            "llm_request_metadata": summary.llm_request_metadata,
        },
    )
    await session.commit()
    return {"conversation_id": conversation.id, "summary": summary.model_dump(mode="json")}


@router.get("/conversations/{conversation_id}", response_model=ConversationDetail)
async def get_conversation(conversation_id: str, session: AsyncSession = Depends(get_session)) -> ConversationDetail:
    conversation = await session.get(Conversation, conversation_id)
    if not conversation:
        raise HTTPException(status_code=404, detail="Conversation not found")

    result = await session.execute(select(Message).where(Message.conversation_id == conversation.id).order_by(Message.created_at.asc()))
    messages = result.scalars().all()
    summary_result = await session.execute(select(IntakeSummaryRecord).where(IntakeSummaryRecord.conversation_id == conversation.id).order_by(IntakeSummaryRecord.created_at.desc()))
    summary = summary_result.scalars().first()

    return ConversationDetail(
        id=conversation.id,
        status=conversation.status,
        channel=conversation.channel,
        started_at=conversation.started_at,
        messages=[
            {
                "id": m.id,
                "role": m.role,
                "content": m.content,
                "created_at": m.created_at.isoformat() if m.created_at else None,
                "model_metadata": m.model_metadata,
            }
            for m in messages
        ],
        summary=summary.raw_summary_json if summary else None,
    )

router = APIRouter(prefix="/api/followups", tags=["followups"])


@router.get("/due", response_model=list[FollowUpTaskView])
async def due_followups(session: AsyncSession = Depends(get_session)) -> list[FollowUpTaskView]:
    now = datetime.utcnow()
    result = await session.execute(
        select(FollowUpTask).where(FollowUpTask.status == "scheduled", FollowUpTask.due_at <= now).order_by(FollowUpTask.due_at.asc())
    )
    return list(result.scalars().all())


@router.post("/run")
async def run_followups(session: AsyncSession = Depends(get_session)) -> dict:
    now = datetime.utcnow()
    result = await session.execute(
        select(FollowUpTask).where(FollowUpTask.status == "scheduled", FollowUpTask.due_at <= now).order_by(FollowUpTask.due_at.asc())
    )
    tasks = result.scalars().all()
    sent = 0
    blocked = 0

    for task in tasks:
        last_message_result = await session.execute(
            select(Message).where(Message.conversation_id == task.conversation_id).order_by(Message.created_at.desc())
        )
        latest = last_message_result.scalars().first()
        allowed, reason = should_send_follow_up(now, settings.outbound_kill_switch, latest.content if latest else None)
        if not allowed:
            task.status = "blocked"
            task.stop_reason = reason
            blocked += 1
            await write_audit(
                session,
                actor="system",
                action="followup.blocked",
                entity_type="follow_up_task",
                entity_id=task.id,
                after_state={"reason": reason},
            )
            continue

        task.status = "sent"
        task.sent_at = now
        sent += 1
        await write_audit(
            session,
            actor="system",
            action="followup.sent",
            entity_type="follow_up_task",
            entity_id=task.id,
            after_state={"payload": task.payload, "template_id": task.template_id},
        )

    await session.commit()
    return {"sent": sent, "blocked": blocked}

from __future__ import annotations

from datetime import datetime
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from app.db import get_session
from app.models import Conversation, IntakeSummaryRecord, Message
from app.schemas import ReviewAction
from app.services.audit import write_audit

router = APIRouter(prefix="/api/review", tags=["review"])


@router.get("/queue")
async def review_queue(session: AsyncSession = Depends(get_session)) -> list[dict]:
    result = await session.execute(
        select(IntakeSummaryRecord).where(IntakeSummaryRecord.review_status == "pending").order_by(IntakeSummaryRecord.created_at.asc())
    )
    rows = result.scalars().all()
    return [
        {
            "summary_id": row.id,
            "conversation_id": row.conversation_id,
            "practice_area": row.primary_practice_area,
            "recommendation": row.agent_recommendation,
            "cross_practice_flag": row.cross_practice_flag,
            "created_at": row.created_at.isoformat(),
        }
        for row in rows
    ]


@router.get("/summaries/{summary_id}")
async def get_summary(summary_id: str, session: AsyncSession = Depends(get_session)) -> dict:
    summary = await session.get(IntakeSummaryRecord, summary_id)
    if not summary:
        raise HTTPException(status_code=404, detail="Summary not found")
    result = await session.execute(select(Message).where(Message.conversation_id == summary.conversation_id).order_by(Message.created_at.asc()))
    messages = result.scalars().all()
    return {
        "summary": summary.raw_summary_json,
        "review_status": summary.review_status,
        "transcript": [{"role": m.role, "content": m.content, "created_at": m.created_at.isoformat()} for m in messages],
    }


@router.post("/summaries/{summary_id}/decision")
async def review_decision(summary_id: str, payload: ReviewAction, session: AsyncSession = Depends(get_session)) -> dict:
    summary = await session.get(IntakeSummaryRecord, summary_id)
    if not summary:
        raise HTTPException(status_code=404, detail="Summary not found")

    before = dict(summary.raw_summary_json)
    summary.review_status = payload.decision.value
    summary.reviewer_id = payload.reviewer_id
    summary.reviewed_at = datetime.utcnow()
    if payload.updated_summary:
        summary.raw_summary_json = payload.updated_summary
    if payload.decision.value == "approved":
        summary.final_disposition = "approved_for_next_step"
    elif payload.decision.value == "rejected":
        summary.final_disposition = "rejected_to_engineering_or_human_followup"
    else:
        summary.final_disposition = "modified_for_review"

    conversation = await session.get(Conversation, summary.conversation_id)
    conversation.status = "in_review"

    await write_audit(
        session,
        actor=f"reviewer:{payload.reviewer_id}",
        action=f"review.{payload.decision.value}",
        entity_type="intake_summary",
        entity_id=summary.id,
        before_state=before,
        after_state=summary.raw_summary_json,
    )
    await session.commit()
    return {"status": "ok", "summary_id": summary.id, "review_status": summary.review_status}
